import time import requests from datetime import datetime, timezone from config import ( BOT_TOKEN, GROUP_CHAT_ID, COINS_PER_MESSAGE, DAILY_CLAIM_AMOUNT, STREAK_BONUS_PER_DAY, STREAK_BONUS_CAP, STARTING_BALANCE, ADMIN_ID, WEBAPP_URL, ) from db import get_user, create_user, add_coins, set_claim API = f"https://api.telegram.org/bot{BOT_TOKEN}" def send_message(chat_id, text, reply_to=None, reply_markup=None): payload = {"chat_id": chat_id, "text": text} if reply_to: payload["reply_to_message_id"] = reply_to if reply_markup: payload["reply_markup"] = reply_markup resp = requests.post(f"{API}/sendMessage", json=payload) if resp.status_code != 200: print(f"[blunt-roll] sendMessage failed ({resp.status_code}): {resp.text}") def ensure_user(user_id: str, username: str): user = get_user(user_id) if not user: create_user(user_id, username, STARTING_BALANCE) user = get_user(user_id) return user def handle_chat_message(message): chat_id = str(message["chat"]["id"]) if GROUP_CHAT_ID and chat_id != GROUP_CHAT_ID: # This is almost always the reason coin-per-message "doesn't work" — # the message arrived from a chat_id that doesn't match GROUP_CHAT_ID. print(f"[blunt-roll] ignored message from chat {chat_id} (GROUP_CHAT_ID={GROUP_CHAT_ID})") return user = message.get("from", {}) user_id = str(user.get("id", "")) username = user.get("username") or user.get("first_name", "user") if not user_id or user.get("is_bot"): return ensure_user(user_id, username) add_coins(user_id, COINS_PER_MESSAGE) print(f"[blunt-roll] +{COINS_PER_MESSAGE} coin -> {username} ({user_id})") def handle_claim_command(message): chat_id = message["chat"]["id"] msg_id = message["message_id"] user = message.get("from", {}) user_id = str(user.get("id", "")) username = user.get("username") or user.get("first_name", "user") record = ensure_user(user_id, username) now = datetime.now(timezone.utc) last_claim_str = record.get("last_claim") streak = record.get("streak", 0) or 0 if last_claim_str: last_claim = datetime.fromisoformat(last_claim_str.replace("Z", "+00:00")) hours_since = (now - last_claim).total_seconds() / 3600 if hours_since < 20: send_message(chat_id, "Already claimed today — come back later.", msg_id) return if hours_since <= 48: streak += 1 # claimed within the streak window, keep it going else: streak = 1 # missed a day, streak resets else: streak = 1 bonus = min(streak * STREAK_BONUS_PER_DAY, STREAK_BONUS_CAP) total = DAILY_CLAIM_AMOUNT + bonus new_balance = record["coins"] + total set_claim(user_id, new_balance, streak) send_message( chat_id, f"+{total} coins claimed (streak: {streak} day{'s' if streak != 1 else ''}). " f"Balance: {new_balance}", msg_id, ) def handle_open_command(message): chat_id = message["chat"]["id"] if not WEBAPP_URL: send_message( chat_id, "WEBAPP_URL isn't set yet — add it in Railway's environment variables " "(the https:// link to the hosted HTML file) then try /open again.", ) return send_message( chat_id, "🎲 Tap below to open Blunt Roll", reply_markup={ "inline_keyboard": [[ {"text": "▶️ Open Blunt Roll", "web_app": {"url": WEBAPP_URL}} ]] }, ) def handle_gift_command(message): chat_id = message["chat"]["id"] msg_id = message["message_id"] sender = message.get("from", {}) sender_id = str(sender.get("id", "")) if sender_id != ADMIN_ID: return # not the admin — stay quiet, don't advertise the command parts = message.get("text", "").split() reply = message.get("reply_to_message") target_id = None target_name = None amount = None if reply: # /gift (as a reply to the person you're gifting) target_user = reply.get("from", {}) target_id = str(target_user.get("id", "")) target_name = target_user.get("username") or target_user.get("first_name", "user") if len(parts) >= 2: try: amount = int(parts[1]) except ValueError: amount = None elif len(parts) >= 3: # /gift target_id = parts[1].lstrip("@") target_name = target_id try: amount = int(parts[2]) except ValueError: amount = None if not target_id or amount is None: send_message( chat_id, "Usage: reply to someone's message with `/gift `, " "or use `/gift `.", msg_id, ) return ensure_user(target_id, target_name) add_coins(target_id, amount) send_message(chat_id, f"🎁 Gifted {amount} coins to {target_name}.", msg_id) def poll_updates(offset=None): params = {"timeout": 20} if offset: params["offset"] = offset resp = requests.get(f"{API}/getUpdates", params=params, timeout=25) data = resp.json() if not data.get("ok"): return offset for update in data.get("result", []): offset = update["update_id"] + 1 message = update.get("message") if not message: continue text = message.get("text", "") if text.startswith("/claim"): handle_claim_command(message) elif text.startswith("/open"): handle_open_command(message) elif text.startswith("/gift"): handle_gift_command(message) else: handle_chat_message(message) return offset def main(): print("[blunt-roll] bot starting up") print(f"[blunt-roll] GROUP_CHAT_ID={GROUP_CHAT_ID!r} WEBAPP_URL={WEBAPP_URL!r} ADMIN_ID={ADMIN_ID!r}") offset = None while True: try: offset = poll_updates(offset) except Exception as e: print(f"[blunt-roll] error: {e}") time.sleep(5) if __name__ == "__main__": main()
